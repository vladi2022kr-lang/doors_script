-- Ультра-стабильный загрузчик для Xeno
local success, result = pcall(function()
    return game:HttpGet("https://raw.githubusercontent.com/vladi2022kr-lang/doors_script/main/doors.txt")
end)

if success and result then
    local runSuccess, err = pcall(loadstring(result))
    if not runSuccess then
        warn("[Loader Error]: Ошибка внутри самого скрипта -> " .. tostring(err))
    end
else
    warn("[Loader Error]: Не удалось скачать скрипт по ссылке (проверь интернет или Raw-ссылку)")
end
