pcall(function()
    local oldNameCall
    oldNameCall = hookmetamethod(game, "__namecall", function(self, ...)
        local method = getnamecallmethod()
        if (method == "Kick" or method == "Ban") and self == game.Players.LocalPlayer then
            return
        end
        return oldNameCall(self, ...)
    end)
end)

local function getRoomsFolder()
    return Workspace:FindFirstChild("Rooms") or Workspace:FindFirstChild("CurrentRooms") or Workspace:FindFirstChild("Floor")
end

AdminTab:CreateSection("Item Spawner")

local function giveItem(itemName)
    pcall(function()
        local itemsFolder = ReplicatedStorage:FindFirstChild("Items") or ReplicatedStorage:FindFirstChild("ToolModules") or ReplicatedStorage
        local itemTemplate = itemsFolder:FindFirstChild(itemName, true)
        
        if itemTemplate then
            itemTemplate:Clone().Parent = localPlayer.Backpack
        else
            local tool = Instance.new("Tool")
            tool.Name = itemName
            tool.RequiresHandle = true
            local handle = Instance.new("Part")
            handle.Name = "Handle"
            handle.Size = Vector3.new(1, 1, 1)
            handle.Parent = tool
            tool.Parent = localPlayer.Backpack
        end
        Rayfield:Notify({Title = "Admin", Content = "Given: " .. itemName, Duration = 2})
    end)
end

local function safeMove(targetPos, hum, char)
    local path = game:GetService("PathfindingService"):CreatePath({ AgentRadius = 2, AgentHeight = 5, AgentCanJump = true })
    local success = pcall(function() path:ComputeAsync(char.HumanoidRootPart.Position, targetPos) end)
    
    if success and path.Status == Enum.PathStatus.Success then
        for _, wp in ipairs(path:GetWaypoints()) do
            if not getgenv().DoorsHub.AutoPlay then break end
            hum:MoveTo(wp.Position)
            if wp.Action == Enum.PathWaypointAction.Jump then hum.Jump = true end
            path.Reached:Wait()
        end
    else
        hum:MoveTo(targetPos)
    end
end

task.spawn(function()
    while task.wait(0.2) do
        if getgenv().DoorsHub.AutoPlay then
            pcall(function()
                local char = localPlayer.Character
                if not char or not char:FindFirstChild("HumanoidRootPart") then return end
                local hum = char:FindFirstChild("Humanoid")
                hum.WalkSpeed = getgenv().DoorsHub.Speed

                local rooms = getRoomsFolder()
                if rooms then
                    for _, room in ipairs(rooms:GetChildren()) do
                        local door = room:FindFirstChild("Door")
                        if door then
                            local knob = door:FindFirstChild("Knob", true)
                            if knob then
                                safeMove(knob.Position, hum, char)
                                if getgenv().DoorsHub.AutoOpen and (char.HumanoidRootPart.Position - knob.Position).Magnitude < 9 then
                                    fireproximityprompt(knob:FindFirstChildOfClass("ProximityPrompt"))
                                    task.wait(0.3)
                                end
                            end
                        end
                    end
                end
            end)
        end
    end
end)
