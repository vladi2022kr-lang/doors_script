-- Защита от киков сервером
pcall(function()
    local oldNameCall
    oldNameCall = hookmetamethod(game, "__namecall", function(self, ...)
        local method = getnamecallmethod()
        if (method == "Kick" or method == "Ban") and self == game.Players.LocalPlayer then
            return
        end
        return oldNameCall(self, ...)
    end)
end)

-- Безопасная загрузка UI с защитой от сбоев
local successLoad, Rayfield = pcall(function()
    return loadstring(game:HttpGet('https://sirius.menu/rayfield'))()
end)

if not successLoad or not Rayfield then
    warn("[Script Error] Не удалось загрузить Rayfield UI. Проверьте интернет или блокировку соединения.")
    return
end

local Window = Rayfield:CreateWindow({
   Name = "DOORS | Universal Hub",
   LoadingTitle = "Запуск системы...",
   LoadingSubtitle = "Проверка этажа...",
   ConfigurationSaving = { Enabled = false },
   KeySystem = false,
})

Rayfield:Notify({
   Title = "Успех",
   Content = "Скрипт успешно инициализирован.",
   Duration = 3
})

local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local UserInputService = game:GetService("UserInputService")
local CoreGui = game:GetService("CoreGui")
local localPlayer = Players.LocalPlayer

-- Настройка поверх интерфейса паузы (Esc)
pcall(function()
    if Rayfield.CoreGui then
        Rayfield.CoreGui.DisplayOrder = 999999
        Rayfield.CoreGui.ResetOnSpawn = false
    end
end)

-- Открытие/закрытие на Правый Шифт
UserInputService.InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.RightShift then
        pcall(function()
            local mainUI = CoreGui:FindFirstChild("Rayfield") or localPlayer.PlayerGui:FindFirstChild("Rayfield")
            if mainUI then
                local container = mainUI:FindFirstChild("Main")
                if container then
                    container.Visible = not container.Visible
                end
            end
        end)
    end
end)

getgenv().DoorsHub = {
    AutoPlay = false,
    AutoOpen = true,
    Speed = 18
}

-- Универсальная функция поиска папки комнат (поддерживает Hotel, Rooms и другие этажи)
local function getRoomsFolder()
    local currentRooms = Workspace:FindFirstChild("Rooms") 
        or Workspace:FindFirstChild("CurrentRooms") 
        or Workspace:FindFirstChild("Floor")
    return currentRooms
end

AdminTab:CreateSection("Выдача предметов")

local function giveItem(itemName)
    pcall(function()
        local itemsFolder = ReplicatedStorage:FindFirstChild("Items") or ReplicatedStorage:FindFirstChild("ToolModules") or ReplicatedStorage
        local itemTemplate = itemsFolder:FindFirstChild(itemName, true)
        
        if itemTemplate then
            itemTemplate:Clone().Parent = localPlayer.Backpack
        else
            local tool = Instance.new("Tool")
            tool.Name = itemName
            tool.RequiresHandle = true
            local handle = Instance.new("Part")
            handle.Name = "Handle"
            handle.Size = Vector3.new(1, 1, 1)
            handle.Parent = tool
            tool.Parent = localPlayer.Backpack
        end
        Rayfield:Notify({Title = "Предмет", Content = "Выдан: " .. itemName, Duration = 2})
    end)
end

AdminTab:CreateButton({
   Name = "Дать: Крест (Crucifix)",
   Callback = function() giveItem("Crucifix") end,
})

AdminTab:CreateButton({
   Name = "Дать: Отмычка (SkeletonKey)",
   Callback = function() giveItem("SkeletonKey") end,
})

-- Безопасный главный цикл без крашей
task.spawn(function()
    while task.wait(0.2) do
        if getgenv().DoorsHub.AutoPlay then
            pcall(function()
                local char = localPlayer.Character
                if not char or not char:FindFirstChild("HumanoidRootPart") then return end
                local hum = char:FindFirstChild("Humanoid")
                hum.WalkSpeed = getgenv().DoorsHub.Speed

                local rooms = getRoomsFolder()
                if rooms then
                    for _, room in ipairs(rooms:GetChildren()) do
                        local door = room:FindFirstChild("Door")
                        if door then
                            local knob = door:FindFirstChild("Knob", true)
                            if knob then
                                hum:MoveTo(knob.Position)
                                if getgenv().DoorsHub.AutoOpen and (char.HumanoidRootPart.Position - knob.Position).Magnitude < 9 then
                                    fireproximityprompt(knob:FindFirstChildOfClass("ProximityPrompt"))
                                    task.wait(0.3)
                                end
                            end
                        end
                    end
                end
            end)
        end
    end
end)
